matches = []
for line in RAW.strip().splitlines():
    date, stage, entries = line.split('|')
    for entry in entries.split(';'):
        home, away, hg, ag = entry.split()
        matches.append(dict(season='2023-24', date=date, stage=stage,
                            home_team=TEAMS[home], away_team=TEAMS[away],
                            home_goals=int(hg), away_goals=int(ag),
                            source='OpenFootball cl.txt (verified transcription)'))
df = pd.DataFrame(matches)
assert len(df) == 125, len(df)
assert df['stage'].value_counts().to_dict() == {'Group': 96, 'Knockout': 29}
assert len(set(df.home_team) | set(df.away_team)) == 32
assert not df.duplicated(['date', 'home_team', 'away_team']).any()
df.to_csv(ROOT / 'data/matches_2023_24.csv', index=False)
